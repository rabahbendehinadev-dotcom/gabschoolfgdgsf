أريد إعادة تصميم تجربة تثبيت التطبيق على iPhone بالكامل لتصبح احترافية (Premium UX) وتشبه جودة تطبيقات Apple.

لقد أرفقت صورة مرجعية، أريد الاعتماد عليها وتنفيذ نفس الفكرة تقريباً.

━━━━━━━━━━━━━━━━━━━━━━
1. عند الضغط على زر "تثبيت التطبيق"
━━━━━━━━━━━━━━━━━━━━━━

إذا كان المستخدم على iPhone:

❌ لا تفتح Safari مباشرة.
❌ لا تنتقل مباشرة إلى أي صفحة.

بدلاً من ذلك افتح Full Screen Modal داخل الموقع يحتوي على دليل التثبيت.

━━━━━━━━━━━━━━━━━━━━━━
2. تصميم الدليل
━━━━━━━━━━━━━━━━━━━━━━

لا أريد أي رسومات كرتونية أو أي Illustrations.

أريد استعمال Screenshots حقيقية من iPhone فقط.

في كل خطوة:

✅ لقطة شاشة حقيقية من Safari.
✅ دائرة برتقالية حول الزر المطلوب.
✅ سهم احترافي يشير إلى الزر.
✅ Zoom على الزر حتى يكون واضحاً.
✅ Blur خفيف للخلفية.
✅ الصورة كبيرة وتشغل أغلب الشاشة.
✅ Animation بسيط لليد وهي تضغط على الزر.
✅ Transition ناعم بين الخطوات.

أريد أن تبدو التجربة وكأنها جزء من نظام iOS.

━━━━━━━━━━━━━━━━━━━━━━
3. الخطوات
━━━━━━━━━━━━━━━━━━━━━━

الخطوة الأولى

العنوان:
اضغط على زر المشاركة

الوصف:
من أسفل متصفح Safari اضغط على زر المشاركة (Share).

━━━━━━━━━━━━━━━━━━━━━━

الخطوة الثانية

العنوان:
اضغط على "En voir plus"

الوصف:
إذا لم تجد "Sur l'écran d'accueil"، اضغط على
"En voir plus"
لعرض جميع الخيارات.

━━━━━━━━━━━━━━━━━━━━━━

الخطوة الثالثة

العنوان:
اختر "Sur l'écran d'accueil"

الوصف:
من قائمة الخيارات اضغط على
"Sur l'écran d'accueil".

━━━━━━━━━━━━━━━━━━━━━━

الخطوة الرابعة

العنوان:
اضغط على "Ajouter"

الوصف:
يمكنك تغيير اسم التطبيق إذا أردت، ثم اضغط على
"Ajouter".

━━━━━━━━━━━━━━━━━━━━━━

الخطوة الخامسة

العنوان:
تم تثبيت التطبيق بنجاح

الوصف:
سيظهر تطبيق GAB School على الشاشة الرئيسية ويمكنك تشغيله مثل أي تطبيق عادي.

━━━━━━━━━━━━━━━━━━━━━━
4. تصميم الـ Modal
━━━━━━━━━━━━━━━━━━━━━━

- Full Screen.
- خلفية Blur.
- زر X كبير للإغلاق.
- إمكانية السحب بين الخطوات.
- مؤشر يوضح رقم الخطوة.
- إمكانية التكبير (Pinch To Zoom).
- إمكانية تحريك الصورة.
- لا يتم قص أي جزء من الصور.

━━━━━━━━━━━━━━━━━━━━━━
5. بعد الانتهاء
━━━━━━━━━━━━━━━━━━━━━━

بعد آخر خطوة يظهر زر كبير:

"فتح Safari"

وعند الضغط عليه فقط يتم فتح الموقع داخل Safari.

أما إذا كان المستخدم داخل Safari أصلاً:

لا تعرض زر "فتح Safari".

اعرض فقط:

"فهمت، متابعة"

ثم أغلق الدليل.

━━━━━━━━━━━━━━━━━━━━━━
6. تذكر المستخدم
━━━━━━━━━━━━━━━━━━━━━━

إذا شاهد المستخدم الدليل مرة واحدة:

لا تعرضه في كل مرة.

احفظ ذلك في LocalStorage.

وأضف داخل الإعدادات:

"إعادة عرض دليل تثبيت iPhone"

━━━━━━━━━━━━━━━━━━━━━━
7. إذا كان المستخدم داخل تطبيقات التواصل
━━━━━━━━━━━━━━━━━━━━━━

إذا كان الرابط مفتوحاً داخل:

Instagram
Facebook
Messenger
TikTok
Snapchat
Telegram
أو أي In-App Browser

اعرض رسالة واضحة:

"لإكمال تثبيت التطبيق يجب فتح الموقع داخل Safari."

ثم اعرض زرين:

- نسخ الرابط
- فتح في Safari (إذا كان ذلك ممكناً)

━━━━━━━━━━━━━━━━━━━━━━
8. الجودة المطلوبة
━━━━━━━━━━━━━━━━━━━━━━

أريد تجربة Premium بنسبة 100%.

لا أريد أي رسومات أو أيقونات كرتونية.

أريد Screenshots حقيقية تماماً مثل الصورة المرفقة.

كل الأسهم والدوائر والـ Zoom تكون احترافية.

الواجهة يجب أن تبدو وكأن Apple نفسها صممتها.

ولا تغيّر نظام الـ PWA الحالي، فقط حسّن تجربة المستخدم الخاصة بتثبيت التطبيق.
